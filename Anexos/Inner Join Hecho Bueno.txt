SELECT DISTINCT
    DIM_TERRITORIO.Id AS "ID_TERRITORIO",
    DIM_CLIENTE.Id AS "ID_CLIENTE",
    DIM_EMPLEADO.Id AS "ID_EMPLEADO",
    DIM_PRODUCTO.Id AS "ID_PRODUCTO",
    DIM_TIEMPO.TiempoID AS "ID_TIEMPO",
    DETALLE_VENTAS_REL.Cantidad AS "CANTIDAD",
    VENTAS_REL.Fecha AS "FECHA",
    VENTAS_REL.FechaEnvio AS "FECHA _ENVIO",
    DETALLE_VENTAS_REL.PrecioUnitario * DETALLE_VENTAS_REL.Cantidad AS "TOTAL_VENTAS"

FROM BikeZ.dbo.Almacenes AS ALMACEN_REL
INNER JOIN BikeZ.dbo.Territorios AS TERRITORIO_REL ON (ALMACEN_REL.TerritorioID = TERRITORIO_REL.TerritorioID)
INNER JOIN BikeZ.dbo.Ventas AS VENTAS_REL ON (VENTAS_REL.TerritorioID = TERRITORIO_REL.TerritorioID)
INNER JOIN BikeZ.dbo.Empleados AS EMPLEADOS_REL ON (EMPLEADOS_REL.TerritorioID = TERRITORIO_REL.TerritorioID)
INNER JOIN BikeZ.dbo.Paises AS PAIS_REL ON (TERRITORIO_REL.PaisID = PAIS_REL.PaisID)
INNER JOIN BikeZ.dbo.Clientes AS CLIENTES_REL ON (CLIENTES_REL.ClienteID = VENTAS_REL.ClienteID)
INNER JOIN BikeZ.dbo.DetalleVentas AS DETALLE_VENTAS_REL ON (VENTAS_REL.VentaID = DETALLE_VENTAS_REL.VentaID)
INNER JOIN BikeZ.dbo.Productos AS PRODUCTOS_REL ON (DETALLE_VENTAS_REL.ProductoID = PRODUCTOS_REL.ProductoID)
INNER JOIN BikeZ.dbo.Categorias AS CATEGORIA_REL ON (PRODUCTOS_REL.CategoriaID = CATEGORIA_REL.CategoriaID)

INNER JOIN BikeZ_DW.dbo.DIM_TERRITORIO AS DIM_TERRITORIO ON (DIM_TERRITORIO.TerritorioID = TERRITORIO_REL.TerritorioID)
INNER JOIN BikeZ_DW.dbo.DIM_PRODUCTO AS DIM_PRODUCTO ON (DIM_PRODUCTO.ProductoID = PRODUCTOS_REL.ProductoID)
INNER JOIN BikeZ_DW.dbo.DIM_CLIENTE AS DIM_CLIENTE ON (DIM_CLIENTE.ClienteID = CLIENTES_REL.ClienteID)
INNER JOIN BikeZ_DW.dbo.DIM_TIEMPO AS DIM_TIEMPO ON (DIM_TIEMPO.Fecha = VENTAS_REL.fecha)
INNER JOIN Bikez_DW.dbo.DIM_EMPLEADO AS DIM_EMPLEADO ON (DIM_EMPLEADO.EmpleadoID = EMPLEADOS_REL.EmpleadoID)

GROUP BY DIM_TERRITORIO.Id, DIM_CLIENTE.Id, DIM_EMPLEADO.Id, DIM_PRODUCTO.Id, DIM_TIEMPO.TiempoID, DETALLE_VENTAS_REL.Cantidad, VENTAS_REL.Fecha, VENTAS_REL.FechaEnvio, DETALLE_VENTAS_REL.Cantidad, DETALLE_VENTAS_REL.PrecioUnitario
